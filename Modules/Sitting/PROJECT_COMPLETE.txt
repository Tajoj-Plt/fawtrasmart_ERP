🚀 خطوات التشغيل
═══════════════════════════════════════════════════════════════════════

1️⃣ تشغيل Migration
   └─ php artisan migrate

2️⃣ مسح الـ Cache (اختياري)
   ├─ php artisan config:clear
   ├─ php artisan route:clear
   └─ php artisan view:clear

3️⃣ فتح الصفحة
   └─ http://your-domain.com/SMPT/index

4️⃣ إدخال البيانات
   ├─ فعّل: "أرسل البريد بإستعمال حساب SMTP"
   ├─ أدخل: اسم المرسل
   ├─ أدخل: البريد الإلكتروني
   ├─ أدخل: كلمة المرور (App Password لـ Gmail)
   └─ اختياري: الخيارات المتقدمة

5️⃣ الحفظ والاختبار
   ├─ اضغط: زر "حفظ"
   └─ اضغط: "إرسال بريد تجريبي"

📊 قاعدة البيانات
═══════════════════════════════════════════════════════════════════════

الجدول: smtp_settings

┌─────────────────┬──────────────┬──────────────────────┐
│ العمود          │ النوع       │ الوصف               │
├─────────────────┼──────────────┼──────────────────────┤
│ id              │ BIGINT       │ معرف فريد           │
│ user_id         │ BIGINT (FK)  │ معرف المستخدم       │
│ use_smtp        │ BOOLEAN      │ تفعيل SMTP          │
│ sender_name     │ VARCHAR(255) │ اسم المرسل          │
│ email           │ VARCHAR(255) │ البريد الإلكتروني   │
│ password        │ TEXT         │ كلمة المرور (مشفرة) │
│ smtp_username   │ VARCHAR(255) │ اسم مستخدم SMTP    │
│ smtp_host       │ VARCHAR(255) │ خادم SMTP           │
│ smtp_port       │ INT          │ المنفذ (587)        │
│ smtp_encryption │ VARCHAR(255) │ التشفير (tls)       │
│ ssl_required    │ BOOLEAN      │ متطلب SSL           │
│ is_active       │ BOOLEAN      │ حالة التفعيل        │
│ created_at      │ TIMESTAMP    │ تاريخ الإنشاء       │
│ updated_at      │ TIMESTAMP    │ تاريخ التحديث       │
└─────────────────┴──────────────┴──────────────────────┘

العلاقات:
  user_id → users(id) ON DELETE CASCADE
